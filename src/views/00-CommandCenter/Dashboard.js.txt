// File: src/views/02-Operations/Dashboard.js
import React, { useMemo, useState, useEffect, useCallback } from 'react';
import { getDashboardKpis, getSalesChartData } from '../../api/analyticsService';
import { getOnlineDrivers, getInventorySummary, getPlants } from '../../api/operationsService';
import PageTitle from '../../components/shared/PageTitle';
import BarChart from '../../components/charts/BarChart';
import {
  TrendingUp,
  ShoppingCart,
  Truck,
  Users,
  Activity,
  ArrowUpRight,
  AlertCircle,
  Zap,
  ExternalLink,
  RefreshCw,
  Factory,
  Package,
  CheckCircle2,
  Database,
} from 'lucide-react';

// Enhanced Glass Card
const StatCard = ({ title, value, icon: Icon, color, trend, sub }) => (
  <div className="glass-card relative overflow-hidden group hover:border-white/20 transition-all duration-300">
    <div
      className={`absolute -right-6 -top-6 p-8 rounded-full opacity-10 group-hover:opacity-20 transition-all ${color} blur-xl`}
    />
    <div className="relative z-10">
      <div className="flex justify-between items-start">
        <div>
          <p className="text-gray-400 text-xs font-bold uppercase tracking-wider">{title}</p>
          <h3 className="text-3xl font-bold text-white mt-2">{value}</h3>
          {sub ? <p className="text-[11px] text-gray-500 mt-2">{sub}</p> : null}
        </div>
        <div className={`p-3 rounded-xl bg-white/5 border border-white/10 ${color.replace('bg-', 'text-')}`}>
          <Icon size={24} />
        </div>
      </div>
      {trend ? (
        <div className="mt-4 flex items-center text-xs font-medium text-green-400">
          <ArrowUpRight size={14} className="mr-1" />
          {trend}
        </div>
      ) : null}
    </div>
  </div>
);

const QuickAction = ({ label, icon: Icon, view, color, setActiveView }) => (
  <button
    onClick={() => setActiveView?.(view)}
    className="flex items-center p-4 rounded-xl bg-white/5 border border-white/5 hover:bg-white/10 hover:border-white/20 transition-all group w-full text-left"
  >
    <div className={`p-2 rounded-lg mr-4 ${color} text-white`}>
      <Icon size={20} />
    </div>
    <div>
      <h4 className="text-sm font-bold text-white group-hover:text-blue-300 transition-colors">{label}</h4>
      <p className="text-[10px] text-gray-500">Jump to module</p>
    </div>
    <ExternalLink
      size={14}
      className="ml-auto text-gray-600 group-hover:text-white opacity-0 group-hover:opacity-100 transition-all"
    />
  </button>
);

const HealthRow = ({ label, value, tone = 'neutral' }) => {
  const toneStyles =
    tone === 'good'
      ? 'bg-green-500/10 border-green-500/20 text-green-300'
      : tone === 'warn'
      ? 'bg-yellow-500/10 border-yellow-500/20 text-yellow-300'
      : tone === 'bad'
      ? 'bg-red-500/10 border-red-500/20 text-red-300'
      : 'bg-white/5 border-white/10 text-gray-300';

  return (
    <div className={`text-xs p-3 rounded-xl border ${toneStyles} flex justify-between items-center`}>
      <span className="font-semibold text-white/90">{label}</span>
      <span className="font-mono">{value}</span>
    </div>
  );
};

const safeNum = (v) => (Number.isFinite(Number(v)) ? Number(v) : 0);

export default function Dashboard({ setActiveView }) {
  const [stats, setStats] = useState({
    totalRevenue: 0,
    totalKgSold: 0,
    activeDeliveries: 0,
    onlineDrivers: 0,
    breakdown: {
      deliveryRecognizedRevenue: 0,
      posRevenue: 0,
      otherRevenue: 0,
    },
    gl: {
      enabled: false,
      source: null,
    },
  });

  const [chartData, setChartData] = useState([]);
  const [loading, setLoading] = useState(true);
  const [lastSyncAt, setLastSyncAt] = useState(null);

  // Filters (aligned with analytics controller)
  const [branches, setBranches] = useState([]);
  const [branchId, setBranchId] = useState('');
  const [period, setPeriod] = useState('monthly');
  const [customStart, setCustomStart] = useState('');
  const [customEnd, setCustomEnd] = useState('');

  // Ops/Health signals
  const [inventory, setInventory] = useState(null);
  const [plants, setPlants] = useState([]);
  const [partialErrors, setPartialErrors] = useState([]);

  // Backend period returned by analytics (actual range used)
  const [backendPeriod, setBackendPeriod] = useState(null);

  const canFetchCustom = period !== 'custom' || (customStart && customEnd);

  const stockPercent = useMemo(() => {
    const cap = safeNum(inventory?.totalCapacity ?? inventory?.totalStockedKg ?? inventory?.capacityKg ?? 0);
    const cur = safeNum(inventory?.currentStock ?? inventory?.currentBulkLpgKg ?? inventory?.currentStockKg ?? 0);
    if (!cap || cap <= 0) return 0;
    return Math.max(0, Math.min(100, (cur / cap) * 100));
  }, [inventory]);

  const plantDownCount = useMemo(
    () => plants.filter((pl) => pl?.status && String(pl.status).toLowerCase() !== 'operational').length,
    [plants]
  );

  const stockTone = stockPercent < 20 ? 'bad' : stockPercent < 35 ? 'warn' : 'good';
  const plantTone = plantDownCount > 0 ? 'warn' : 'good';
  const fleetTone = safeNum(stats.onlineDrivers) === 0 ? 'warn' : 'good';

  const branchLabel = useMemo(() => {
    if (!branchId) return 'All Branches';
    const found = branches.find((b) => String(b.id) === String(branchId));
    return found?.name || `Branch ${branchId}`;
  }, [branchId, branches]);

  const backendPeriodLabel = useMemo(() => {
    const s = backendPeriod?.start ? new Date(backendPeriod.start) : null;
    const e = backendPeriod?.end ? new Date(backendPeriod.end) : null;
    if (!s || !e || Number.isNaN(s.getTime()) || Number.isNaN(e.getTime())) return null;
    return `${s.toISOString().slice(0, 10)} → ${e.toISOString().slice(0, 10)}`;
  }, [backendPeriod]);

  const periodLabel = useMemo(() => {
    if (period === 'custom') return `${customStart || '—'} → ${customEnd || '—'}`;
    if (period === 'monthly') return 'This Month';
    if (period === 'quarterly') return 'This Quarter';
    if (period === 'yearly') return 'This Year';
    return period;
  }, [period, customStart, customEnd]);

  const fetchBranches = useCallback(async () => {
    try {
      const res = await getPlants?.();
      const normalized = Array.isArray(res)
        ? res
            .map((p) => ({
              id: p?.id || p?._id || p?.value,
              name: p?.name || p?.plantName || p?.label || 'Unnamed Branch',
              status: p?.status,
            }))
            .filter((p) => p.id)
        : [];

      setBranches(normalized);
      setPlants(Array.isArray(res) ? res : []);
    } catch (e) {
      console.error('Failed to load plants/branches:', e);
      setBranches([]);
      setPlants([]);
    }
  }, []);

  const fetchData = useCallback(async () => {
    if (period === 'custom' && (!customStart || !customEnd)) {
      setLoading(false);
      return;
    }

    setLoading(true);
    setPartialErrors([]);

    try {
      const from = period === 'custom' ? customStart : undefined;
      const to = period === 'custom' ? customEnd : undefined;
      const selectedBranch = branchId || undefined;

      const analyticsArgs = [period, from, to, selectedBranch];

      // Use allSettled so one failure doesn't crash the whole dashboard
      const results = await Promise.allSettled([
        getDashboardKpis(...analyticsArgs),
        getOnlineDrivers?.(),
        getSalesChartData(period, from, to, selectedBranch),
        getInventorySummary?.(selectedBranch),
        // keep plants fresh too (some UIs update statuses)
        getPlants?.(),
      ]);

      // 1) KPIs
      const kpisRaw =
        results[0].status === 'fulfilled'
          ? (results[0].value?.data ?? results[0].value ?? {})
          : { totalRevenue: 0, totalKgSold: 0, activeDeliveries: 0, breakdown: {} };

      // 2) Drivers
      const driversVal = results[1].status === 'fulfilled' ? (results[1].value?.data ?? results[1].value) : [];
      const drivers = Array.isArray(driversVal) ? driversVal : [];

      // 3) Sales Chart
      const sales = results[2].status === 'fulfilled' ? (results[2].value?.data ?? results[2].value) : null;

      // 4) Inventory Summary (optional)
      const inv = results[3]?.status === 'fulfilled' ? (results[3].value?.data ?? results[3].value) : null;

      // 5) Plant health (optional)
      const p = results[4]?.status === 'fulfilled' ? (results[4].value?.data ?? results[4].value) : [];

      // Collect partial failures (for debugging visibility)
      const errs = [];
      results.forEach((r, idx) => {
        if (r.status === 'rejected') errs.push({ idx, message: r.reason?.message || String(r.reason) });
      });
      setPartialErrors(errs);

      setStats({
        totalRevenue: safeNum(kpisRaw?.totalRevenue),
        totalKgSold: safeNum(kpisRaw?.totalKgSold),
        activeDeliveries: safeNum(kpisRaw?.activeDeliveries),
        onlineDrivers: drivers.length,
        breakdown: {
          deliveryRecognizedRevenue: safeNum(kpisRaw?.breakdown?.deliveryRecognizedRevenue),
          posRevenue: safeNum(kpisRaw?.breakdown?.posRevenue),
          otherRevenue: safeNum(kpisRaw?.breakdown?.otherRevenue),
        },
        gl: {
          enabled: Boolean(kpisRaw?.gl?.enabled),
          source: kpisRaw?.gl?.source || null,
        },
      });

      setBackendPeriod(kpisRaw?.period || null);
      setInventory(inv);
      setPlants(Array.isArray(p) ? p : []);

      // Support multiple backend shapes:
      // - { breakdown: [{label, revenue}] }
      // - array of {_id:{year,month}, totalRevenue}
      // - array of {label, value}
      // - { rows: [...] } (some controllers return wrapped rows)
      const salesRows = Array.isArray(sales?.rows) ? sales.rows : sales;

      if (sales?.breakdown && Array.isArray(sales.breakdown)) {
        setChartData(
          sales.breakdown.map((item) => ({
            label: String(item?.label ?? 'N/A'),
            value: safeNum(item?.revenue ?? item?.value),
          }))
        );
      } else if (Array.isArray(salesRows)) {
        const mapped = salesRows.map((item) => {
          if (item?.label != null) {
            return {
              label: String(item.label),
              value: safeNum(item?.revenue ?? item?.value ?? item?.totalRevenue),
            };
          }

          if (item?._id && typeof item._id === 'object' && item._id.year && item._id.month) {
            const y = item._id.year;
            const m = String(item._id.month).padStart(2, '0');
            return { label: `${y}-${m}`, value: safeNum(item.totalRevenue) };
          }

          return {
            label: String(item?._id ?? 'N/A'),
            value: safeNum(item?.totalRevenue ?? item?.value),
          };
        });

        setChartData(mapped);
      } else {
        setChartData([]);
      }

      setLastSyncAt(new Date());
    } catch (error) {
      console.error('Dashboard Sync Failed', error);
      setPartialErrors([{ idx: -1, message: error?.message || 'Dashboard sync failed' }]);
    } finally {
      setLoading(false);
    }
  }, [period, customStart, customEnd, branchId]);

  useEffect(() => {
    fetchBranches();
  }, [fetchBranches]);

  useEffect(() => {
    fetchData();

    const interval = setInterval(() => {
      fetchData();
    }, 30000);

    return () => clearInterval(interval);
  }, [fetchData]);

  return (
    <div className="space-y-8 animate-in fade-in duration-500">
      {/* Header Area */}
      <div className="flex flex-col gap-4 lg:flex-row lg:justify-between lg:items-end">
        <PageTitle title="Command Center" subtitle="Real-time Operational Intelligence" />

        <div className="flex flex-wrap items-center gap-3">
          {lastSyncAt ? (
            <div className="hidden md:flex items-center px-3 py-1.5 rounded-full bg-white/5 border border-white/10 text-gray-300 text-xs font-mono">
              Last sync: {lastSyncAt.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
            </div>
          ) : null}

          <button
            onClick={fetchData}
            className="glass-button px-3 py-1.5 flex items-center text-xs font-semibold"
            title="Refresh"
          >
            <RefreshCw size={14} className={`mr-2 ${loading ? 'animate-spin' : ''}`} />
            Refresh
          </button>

          <div className="flex items-center space-x-2 px-3 py-1.5 rounded-full bg-green-500/10 border border-green-500/20 text-green-400 text-xs font-mono">
            <span className="relative flex h-2 w-2">
              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-green-400 opacity-75" />
              <span className="relative inline-flex rounded-full h-2 w-2 bg-green-500" />
            </span>
            <span>SYSTEM ONLINE</span>
          </div>
        </div>
      </div>

      {/* Filters */}
      <div className="glass-card">
        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-5 gap-3 items-end">
          <div>
            <label className="text-xs text-gray-400 block mb-1">Branch</label>
            <select
              className="w-full px-3 py-2 rounded-lg bg-black/30 text-white border border-white/10"
              value={branchId}
              onChange={(e) => setBranchId(e.target.value)}
            >
              <option value="">All Branches</option>
              {branches.map((b) => (
                <option key={b.id} value={b.id}>
                  {b.name}
                </option>
              ))}
            </select>
            <p className="text-[10px] text-gray-500 mt-1">Maps to branchId / serviceZoneId</p>
          </div>

          <div>
            <label className="text-xs text-gray-400 block mb-1">Period</label>
            <select
              className="w-full px-3 py-2 rounded-lg bg-black/30 text-white border border-white/10"
              value={period}
              onChange={(e) => setPeriod(e.target.value)}
            >
              <option value="monthly">Monthly</option>
              <option value="quarterly">Quarterly</option>
              <option value="yearly">Yearly</option>
              <option value="custom">Custom</option>
            </select>
          </div>

          {period === 'custom' && (
            <>
              <div>
                <label className="text-xs text-gray-400 block mb-1">Start</label>
                <input
                  type="date"
                  className="w-full px-3 py-2 rounded-lg bg-black/30 text-white border border-white/10"
                  value={customStart}
                  onChange={(e) => setCustomStart(e.target.value)}
                />
              </div>
              <div>
                <label className="text-xs text-gray-400 block mb-1">End</label>
                <input
                  type="date"
                  className="w-full px-3 py-2 rounded-lg bg-black/30 text-white border border-white/10"
                  value={customEnd}
                  onChange={(e) => setCustomEnd(e.target.value)}
                />
              </div>
            </>
          )}

          <div className="text-xs text-gray-400 xl:text-right">
            <div>
              Scope: <span className="text-white">{branchLabel}</span>
            </div>
            <div>
              UI Period: <span className="text-white">{periodLabel}</span>
            </div>
            {backendPeriodLabel ? (
              <div className="text-[10px] text-gray-500 mt-1">
                Backend range: <span className="text-gray-300">{backendPeriodLabel}</span>
              </div>
            ) : null}
            <div className="text-[10px] mt-1">
              Source:{' '}
              <span className={`inline-flex items-center gap-1 ${stats.gl.enabled ? 'text-emerald-300' : 'text-gray-400'}`}>
                <Database size={12} />
                {stats.gl.enabled ? `GL${stats.gl.source ? ` • ${stats.gl.source}` : ''}` : 'Analytics'}
              </span>
            </div>
          </div>
        </div>

        {period === 'custom' && !canFetchCustom ? (
          <div className="mt-3 text-xs text-yellow-300 bg-yellow-500/10 border border-yellow-500/20 rounded-xl p-3">
            Select both custom start and end dates to load analytics.
          </div>
        ) : null}
      </div>

      {/* KPI Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
        <StatCard
          title="Total Revenue"
          value={loading ? '...' : `₦${safeNum(stats.totalRevenue).toLocaleString()}`}
          icon={TrendingUp}
          color="bg-green-500"
          trend="Recognized revenue"
          sub="Paid delivery + approved POS"
        />
        <StatCard
          title="LPG Output"
          value={loading ? '...' : `${safeNum(stats.totalKgSold).toLocaleString()} kg`}
          icon={ShoppingCart}
          color="bg-blue-500"
          trend="Across ledgers"
          sub="Delivery + POS volume"
        />
        <StatCard
          title="Live Deliveries"
          value={loading ? '...' : safeNum(stats.activeDeliveries)}
          icon={Truck}
          color="bg-purple-500"
          sub="Orders in-progress"
        />
        <StatCard
          title="Active Fleet"
          value={loading ? '...' : safeNum(stats.onlineDrivers)}
          icon={Users}
          color="bg-orange-500"
          sub="Drivers online"
        />
      </div>

      {/* Revenue Recognition Diagnostics */}
      <div className="glass-card border border-white/10">
        <div className="flex items-center justify-between gap-3 mb-3">
          <h3 className="text-sm font-bold text-gray-300 uppercase tracking-wider flex items-center">
            <CheckCircle2 size={16} className="mr-2 text-emerald-400" />
            Revenue Recognition Snapshot
          </h3>
          <div className="text-[10px] text-gray-500">Aligned to Management Accounts rules</div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
          <div className="p-3 rounded-xl bg-black/20 border border-white/10">
            <div className="text-[11px] text-gray-400">Delivery (Recognized)</div>
            <div className="text-lg font-bold text-white mt-1">
              ₦{safeNum(stats.breakdown?.deliveryRecognizedRevenue).toLocaleString()}
            </div>
            <div className="text-[10px] text-gray-500 mt-1">Delivered + paid only</div>
          </div>
          <div className="p-3 rounded-xl bg-black/20 border border-white/10">
            <div className="text-[11px] text-gray-400">POS Revenue</div>
            <div className="text-lg font-bold text-white mt-1">
              ₦{safeNum(stats.breakdown?.posRevenue).toLocaleString()}
            </div>
            <div className="text-[10px] text-gray-500 mt-1">Approved daily summaries</div>
          </div>
          <div className="p-3 rounded-xl bg-black/20 border border-white/10">
            <div className="text-[11px] text-gray-400">Total Recognized Revenue</div>
            <div className="text-lg font-bold text-white mt-1">₦{safeNum(stats.totalRevenue).toLocaleString()}</div>
            <div className="text-[10px] text-gray-500 mt-1">Used by dashboard KPI</div>
          </div>
        </div>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* Main Chart */}
        <div className="lg:col-span-2 glass-card min-h-[400px]">
          <div className="flex justify-between items-center mb-6">
            <h3 className="text-lg font-bold text-white flex items-center">
              <Activity size={20} className="mr-2 text-blue-400" /> Revenue Trajectory
            </h3>
            <div className="text-[11px] text-gray-500">
              {chartData.length ? `${periodLabel} breakdown` : 'No sales chart data yet'}
            </div>
          </div>
          <BarChart data={chartData} />
        </div>

        {/* Right Column */}
        <div className="space-y-6">
          {/* Quick Actions */}
          <div className="glass-card">
            <h3 className="text-sm font-bold text-gray-400 uppercase tracking-wider mb-4 flex items-center">
              <Zap size={16} className="mr-2 text-yellow-400" /> Quick Actions
            </h3>
            <div className="space-y-3">
              <QuickAction label="Dispatch Order" icon={Truck} view="Logistics" color="bg-blue-600" setActiveView={setActiveView} />
              <QuickAction label="Log Sale (POS)" icon={ShoppingCart} view="DailyLog" color="bg-green-600" setActiveView={setActiveView} />
              <QuickAction label="Inventory Intelligence" icon={Package} view="Inventory" color="bg-slate-700" setActiveView={setActiveView} />
              <QuickAction label="Plant Performance" icon={Factory} view="PlantStatus" color="bg-indigo-700" setActiveView={setActiveView} />
            </div>
          </div>

          {/* Operational Health */}
          <div className="glass-card border-t-4 border-blue-500">
            <h3 className="text-white font-bold mb-3 flex items-center">
              <AlertCircle size={18} className="mr-2 text-blue-400" /> Operational Health
            </h3>

            <div className="space-y-2">
              <HealthRow
                label="Stock Level"
                value={
                  inventory
                    ? `${stockPercent.toFixed(0)}% (${safeNum(
                        inventory.currentStock ?? inventory.currentBulkLpgKg ?? inventory.currentStockKg ?? 0
                      ).toLocaleString()} kg)`
                    : 'N/A'
                }
                tone={inventory ? stockTone : 'neutral'}
              />
              <HealthRow
                label="Plants"
                value={`${plants.length || 0} total • ${plantDownCount} attention`}
                tone={plants.length ? plantTone : 'neutral'}
              />
              <HealthRow label="Fleet Online" value={`${safeNum(stats.onlineDrivers)} drivers`} tone={fleetTone} />
            </div>

            {partialErrors.length > 0 ? (
              <div className="mt-4 text-[11px] text-yellow-300/80 bg-yellow-500/10 border border-yellow-500/20 rounded-xl p-3">
                <div className="font-semibold text-yellow-200 mb-1">Partial data loaded</div>
                <ul className="list-disc list-inside space-y-1">
                  {partialErrors.slice(0, 3).map((e, idx) => (
                    <li key={idx} className="truncate">
                      {e.message}
                    </li>
                  ))}
                </ul>
              </div>
            ) : null}
          </div>

          {/* System Alerts */}
          <div className="glass-card border-t-4 border-red-500">
            <h3 className="text-white font-bold mb-3 flex items-center">
              <AlertCircle size={18} className="mr-2 text-red-500" /> System Alerts
            </h3>
            <div className="space-y-2">
              <div className="text-xs text-gray-400 p-2 bg-white/5 rounded border-l-2 border-yellow-500">
                <span className="text-white font-bold">Low Stock:</span> Main Plant (Lekki) at 15% capacity.
              </div>
              <div className="text-xs text-gray-400 p-2 bg-white/5 rounded border-l-2 border-blue-500">
                <span className="text-white font-bold">Audit:</span> 3 Unreconciled EOD reports pending.
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
