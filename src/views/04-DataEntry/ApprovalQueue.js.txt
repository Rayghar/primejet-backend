// src/views/04-DataEntry/ApprovalQueue.js
import React, { useEffect, useMemo, useState } from 'react';
import { getPendingApprovals, approveSummary, rejectSummary } from '../../api/dataEntryService';
import { formatCurrency, formatDate } from '../../utils/formatters';
import PageTitle from '../../components/shared/PageTitle';
import Notification from '../../components/shared/Notification';
import Button from '../../components/shared/Button';
import { CheckCircle, XCircle, AlertTriangle, Search, RefreshCw } from 'lucide-react';

const safeArr = (v) => (Array.isArray(v) ? v : []);
const safeId = (o) => o?.id || o?._id;
const safeNum = (v, d = 0) => {
  const n = Number(v);
  return Number.isFinite(n) ? n : d;
};

export default function ApprovalQueue() {
  const [summaries, setSummaries] = useState([]);
  const [loading, setLoading] = useState(true);
  const [actingId, setActingId] = useState(null);
  const [search, setSearch] = useState('');
  const [notification, setNotification] = useState({ show: false, message: '', type: 'success' });

  const fetchQueue = async () => {
    setLoading(true);
    try {
      const data = await getPendingApprovals();
      setSummaries(safeArr(data?.summaries || data));
    } catch (e) {
      console.error('Approval Queue Load Error', e);
      setNotification({ show: true, message: 'Failed to load queue', type: 'error' });
      setSummaries([]);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchQueue();
  }, []);

  const filtered = useMemo(() => {
    const q = String(search || '').toLowerCase();
    return summaries.filter((s) => {
      const branch = String(s.branchName || s.branchId || '').toLowerCase();
      const cashier = String(s.cashierName || '').toLowerCase();
      const date = String(s.date || s.createdAt || '').toLowerCase();
      return branch.includes(q) || cashier.includes(q) || date.includes(q);
    });
  }, [summaries, search]);

  const handleAction = async (id, action) => {
    const confirmMsg = action === 'approve' ? 'Approve this report?' : 'Reject this report?';
    if (!window.confirm(confirmMsg)) return;

    setActingId(id);
    try {
      if (action === 'approve') await approveSummary(id);
      else await rejectSummary(id);

      setNotification({ show: true, message: `Report ${action}d successfully`, type: 'success' });
      await fetchQueue();
    } catch (e) {
      console.error('Approval Action Error', e);
      setNotification({ show: true, message: e?.response?.data?.message || 'Action failed', type: 'error' });
    } finally {
      setActingId(null);
    }
  };

  return (
    <div className="space-y-6">
      <Notification notification={notification} setNotification={setNotification} />

      <div className="flex justify-between items-center">
        <PageTitle title="Approval Queue" subtitle="Pending End-of-Day Reports" />

        <div className="flex items-center gap-2">
          <div className="relative">
            <input
              type="text"
              placeholder="Search branch/cashier/date..."
              className="glass-input pl-9 py-2 w-64 text-sm"
              value={search}
              onChange={(e) => setSearch(e.target.value)}
            />
            <Search className="absolute left-3 top-2.5 text-gray-500" size={16} />
          </div>

          <button
            onClick={fetchQueue}
            className="glass-button px-3 py-2 flex items-center text-sm"
            disabled={loading}
            title="Refresh"
          >
            <RefreshCw size={16} className={`mr-2 ${loading ? 'animate-spin' : ''}`} />
            Refresh
          </button>
        </div>
      </div>

      {loading ? (
        <p className="text-center text-gray-500 animate-pulse">Checking for pending reports...</p>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-6">
          {filtered.length === 0 ? (
            <div className="col-span-full text-center py-20 text-gray-500">
              <CheckCircle size={48} className="mx-auto mb-4 text-green-500/50" />
              <p>All caught up! No pending approvals.</p>
            </div>
          ) : (
            filtered.map((sum) => {
              const id = safeId(sum);
              const discrepancy = safeNum(sum?.reconciliation?.discrepancy, 0);

              return (
                <div key={id} className="glass-card border-l-4 border-yellow-500 relative">
                  <div className="flex justify-between items-start mb-4">
                    <div>
                      <h3 className="font-bold text-white text-lg">{sum.branchName || 'Unknown Branch'}</h3>
                      <p className="text-xs text-gray-400">
                        {formatDate(sum.date || sum.createdAt)} • {sum.cashierName || '—'}
                      </p>
                    </div>
                    <span className="bg-yellow-500/20 text-yellow-500 px-2 py-1 rounded text-xs font-bold uppercase">
                      Review
                    </span>
                  </div>

                  <div className="space-y-3 mb-6 bg-black/20 p-4 rounded-xl">
                    <div className="flex justify-between text-sm">
                      <span className="text-gray-400">Total Revenue</span>
                      <span className="text-green-400 font-bold">{formatCurrency(sum.totalRevenue || sum?.totals?.totalRevenue)}</span>
                    </div>
                    <div className="flex justify-between text-sm">
                      <span className="text-gray-400">Total Expenses</span>
                      <span className="text-red-400 font-bold">{formatCurrency(sum.totalExpenses || sum?.totals?.totalExpenses)}</span>
                    </div>
                    <div className="border-t border-white/10 pt-2 flex justify-between text-sm">
                      <span className="text-white">Net Cash</span>
                      <span className="text-white font-bold">{formatCurrency(sum.netCash || sum?.totals?.netCash)}</span>
                    </div>
                  </div>

                  {discrepancy !== 0 && (
                    <div className="flex items-center text-xs text-red-400 mb-4 bg-red-500/10 p-2 rounded">
                      <AlertTriangle size={14} className="mr-2" />
                      Discrepancy: {formatCurrency(discrepancy)}
                    </div>
                  )}

                  <div className="flex gap-2">
                    <Button
                      onClick={() => handleAction(id, 'approve')}
                      className="flex-1 bg-green-600 hover:bg-green-500 text-xs"
                      disabled={actingId === id}
                      icon={CheckCircle}
                    >
                      {actingId === id ? 'Working…' : 'Approve'}
                    </Button>

                    <Button
                      onClick={() => handleAction(id, 'reject')}
                      className="flex-1 bg-red-600/20 hover:bg-red-600/40 text-red-400 border border-red-500/30 text-xs"
                      disabled={actingId === id}
                      icon={XCircle}
                    >
                      {actingId === id ? 'Working…' : 'Reject'}
                    </Button>
                  </div>
                </div>
              );
            })
          )}
        </div>
      )}
    </div>
  );
}
